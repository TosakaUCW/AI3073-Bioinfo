\section*{Problem 2}
On an abstract sequence of length $18$, we are given weighted candidate exons (closed intervals):
\[
    [1,4]:4,\quad [3,6]:5,\quad [3,8]:10,\quad [9,10]:1,\quad
    [8,16]:12,\quad [15,18]:6,\quad [13,16]:7,\quad [7,11]:8.
\]
Find a maximum-weight subset of \emph{non-overlapping} exons. Intervals may touch at endpoints:
if we choose $[L_i,R_i]$ and $[L_j,R_j]$, they must satisfy $R_i\le L_j$ or $R_j\le L_i$.

\paragraph{Model (WIS).}
Let interval $j$ have endpoints $(L_j,R_j)$ and weight $w_j$. Sort by nondecreasing $R$ and define
\[
    p(j)\;=\;\max\{\,i<j:\ R_i\le L_j\,\},\qquad p(j)=0\text{ if none exists.}
\]
The optimal values satisfy
\[
    M[0]=0,\qquad
    M[j]=\max\!\big(M[j-1],\,w_j+M[p(j)]\big)\quad (j\ge 1),
\]
and we take $j$ exactly when $w_j+M[p(j)]>M[j-1]$.

\paragraph{Data preparation.}
Label the intervals:
\[
    \begin{array}{lcl}
        A=[1,4]:4,   & \; & B=[3,6]:5,\; C=[3,8]:10,\; D=[9,10]:1,    \\
        E=[8,16]:12, & \; & F=[15,18]:6,\; G=[13,16]:7,\; H=[7,11]:8.
    \end{array}
\]
Sort by right endpoint:
\[
    A(1,4,4),\; B(3,6,5),\; C(3,8,10),\; D(9,10,1),\; H(7,11,8),\;
    E(8,16,12),\; G(13,16,7),\; F(15,18,6).
\]
With endpoint-touching allowed ($R_i\le L_j$), the compatibility indices are
\[
    p(1)=0,\; p(2)=0,\; p(3)=0,\; p(4)=3,\; p(5)=2,\; p(6)=3,\; p(7)=5,\; p(8)=5.
\]
(Note the only change from the strict case $R_i<L_j$ is $p(6)$ becomes $3$ since $C$ and $E$ can now touch.)

\paragraph{Dynamic programming table.}
Let $w_j$ be the weight of the $j$-th interval in the sorted list.

\medskip
\begin{tabular}{@{}cccccc@{}}
    \toprule
    $j$ & Interval $(L,R,w)$ & $p(j)$ & $w_j+M[p(j)]$ & $M[j-1]$ & $M[j]=\max$ \\
    \midrule
    0   & --                 & --     & --            & --       & $0$         \\
    1   & $(1,4,4)$          & $0$    & $4$           & $0$      & $4$         \\
    2   & $(3,6,5)$          & $0$    & $5$           & $4$      & $5$         \\
    3   & $(3,8,10)$         & $0$    & $10$          & $5$      & $10$        \\
    4   & $(9,10,1)$         & $3$    & $1+10=11$     & $10$     & $11$        \\
    5   & $(7,11,8)$         & $2$    & $8+5=13$      & $11$     & $13$        \\
    6   & $(8,16,12)$        & $3$    & $12+10=22$    & $13$     & $22$        \\
    7   & $(13,16,7)$        & $5$    & $7+13=20$     & $22$     & $22$        \\
    8   & $(15,18,6)$        & $5$    & $6+13=19$     & $22$     & $22$        \\
    \bottomrule
\end{tabular}

\medskip
Thus the optimal value is $M[8]=\boxed{22}$.

\paragraph{Reconstruction (one optimal solution).}
From $j=8$: $M[8]=M[7]$ (do not take $F$).
At $j=7$: $M[7]=M[6]$ (do not take $G$).
At $j=6$: $12+M[3]=22>M[5]=13$ (take $E=[8,16]:12$), jump to $j=p(6)=3$.
At $j=3$: $10+M[0]=10>M[2]=5$ (take $C=[3,8]:10$), jump to $j=p(3)=0$ and stop.

\paragraph{Answer.}
One maximum-weight non-overlapping set is
\[
    \boxed{\{[3,8],[8,16]\}}
\]
with total weight
\[
    \boxed{10+12=22}.
\]
