\section*{Problem 3}
Align
\[
    \texttt{Sequence\_A}=\texttt{GATTCATA}\quad(\text{length }8), \qquad
    \texttt{Sequence\_B}=\texttt{GGATCGA}\quad(\text{length }7),
\]
with scoring: match $=+5$, mismatch $=-2$, indel (gap) $=-6$. Construct the DP matrix and report the optimal (global) alignment score.

\paragraph{Algorithm (global alignment).}
Let $A_1\ldots A_n$ and $B_1\ldots B_m$ be the two sequences ($n=8$, $m=7$).
Define the score matrix $D\in\mathbb{Z}^{(n+1)\times(m+1)}$:
\[
    \begin{aligned}
         & D(0,0)=0,\qquad D(i,0)=i\cdot(-6),\quad D(0,j)=j\cdot(-6), \\
         & D(i,j)=\max\!\Big\{
        \underbrace{D(i\!-\!1,j\!-\!1)+s(A_i,B_j)}_{\text{diag (match/mismatch)}},\
        \underbrace{D(i\!-\!1,j)+(-6)}_{\text{up (gap in $B$)}},\
        \underbrace{D(i,j\!-\!1)+(-6)}_{\text{left (gap in $A$)}}
        \Big\},
    \end{aligned}
\]
where $s(x,y)=+5$ if $x=y$, otherwise $-2$.

\paragraph{Optimal score.}
The optimal global alignment score is
\[
    \boxed{8}.
\]

\paragraph{One optimal alignment (traceback).}
A valid traceback yielding the score $8$ is:
\[
    \begin{array}{@{}c@{}}
        \texttt{G A T T C A T A} \\
        \texttt{G G A T C - G A}
    \end{array}
    \quad\Rightarrow\quad
    5 - 2 - 2 + 5 + 5 - 6 - 2 + 5 \;=\; 8.
\]
